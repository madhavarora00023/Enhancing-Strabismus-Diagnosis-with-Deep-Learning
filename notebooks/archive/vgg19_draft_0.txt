# Build VGGNet-19 model
def build_VGGNet19(input_shape, num_classes):
    inputs = Input(shape=input_shape)

    # Block 1
    x = Conv2D(64, (3, 3), padding='same', activation='relu')(inputs)
    x = Conv2D(64, (3, 3), padding='same', activation='relu')(x)
    x = MaxPooling2D((2, 2), strides=(2, 2))(x)

    # Block 2
    x = Conv2D(128, (3, 3), padding='same', activation='relu')(x)
    x = Conv2D(128, (3, 3), padding='same', activation='relu')(x)
    x = MaxPooling2D((2, 2), strides=(2, 2))(x)

    # Block 3
    x = Conv2D(256, (3, 3), padding='same', activation='relu')(x)
    x = Conv2D(256, (3, 3), padding='same', activation='relu')(x)
    x = Conv2D(256, (3, 3), padding='same', activation='relu')(x)
    x = Conv2D(256, (3, 3), padding='same', activation='relu')(x)
    x = MaxPooling2D((2, 2), strides=(2, 2))(x)

    # Block 4
    x = Conv2D(512, (3, 3), padding='same', activation='relu')(x)
    x = Conv2D(512, (3, 3), padding='same', activation='relu')(x)
    x = Conv2D(512, (3, 3), padding='same', activation='relu')(x)
    x = Conv2D(512, (3, 3), padding='same', activation='relu')(x)
    x = MaxPooling2D((2, 2), strides=(2, 2))(x)

    # Block 5
    x = Conv2D(512, (3, 3), padding='same', activation='relu')(x)
    x = Conv2D(512, (3, 3), padding='same', activation='relu')(x)
    x = Conv2D(512, (3, 3), padding='same', activation='relu')(x)
    x = Conv2D(512, (3, 3), padding='same', activation='relu')(x)
    x = MaxPooling2D((2, 2), strides=(2, 2))(x)

    # Fully Connected Layers
    x = Flatten()(x)
    x = Dense(4096, activation='relu')(x)
    x = Dense(4096, activation='relu')(x)
    outputs = Dense(num_classes, activation='softmax')(x)

    model = Model(inputs=inputs, outputs=outputs)
    return model

# Function to calculate metrics (accuracy, precision, recall, F1-score)
def calculate_metrics(model, data_generator):
    all_true_labels = []
    all_pred_labels = []

    for batch_images, batch_labels in data_generator:
        preds = model.predict(batch_images, verbose=0)
        all_true_labels.extend(np.argmax(batch_labels, axis=-1))
        all_pred_labels.extend(np.argmax(preds, axis=-1))

        # Stop early to prevent looping indefinitely in generator
        if len(all_true_labels) >= data_generator.samples:
            break

    accuracy = accuracy_score(all_true_labels, all_pred_labels)
    precision = precision_score(all_true_labels, all_pred_labels, average='weighted', zero_division=1)
    recall = recall_score(all_true_labels, all_pred_labels, average='weighted', zero_division=1)
    f1 = f1_score(all_true_labels, all_pred_labels, average='weighted', zero_division=1)

    return accuracy, precision, recall, f1

# Custom callback to compute accuracy, precision, recall, and F1-score at the end of each epoch
class MetricsCallback(Callback):
    def __init__(self, val_data):
        super().__init__()
        self.val_data = val_data  # Store validation data

    def on_epoch_end(self, epoch, logs=None):
        accuracy, precision, recall, f1 = calculate_metrics(self.model, self.val_data)

        print(f"\nEpoch {epoch+1} Metrics - Accuracy: {accuracy:.4f}, Precision: {precision:.4f}, Recall: {recall:.4f}, F1 Score: {f1:.4f}")

        # Store the values in logs dictionary for use in training history
        logs['val_accuracy'] = accuracy
        logs['val_precision'] = precision
        logs['val_recall'] = recall
        logs['val_f1'] = f1

# Get class count
num_classes = train_generator.num_classes

# Build the VGGNet-19 model
model = build_VGGNet19((224, 224, 3), num_classes)

# Compile the model
model.compile(optimizer='adam', loss='categorical_crossentropy', metrics=['accuracy'])

# Train the model with the additional metrics callback
history = model.fit(
    train_generator,
    validation_data=val_generator,
    epochs=30,
    callbacks=[MetricsCallback(val_generator)]
)
